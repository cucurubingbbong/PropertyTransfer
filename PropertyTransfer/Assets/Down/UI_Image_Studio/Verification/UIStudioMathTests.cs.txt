// Optional Unity EditMode test examples for UIStudioMath.
// Rename to .cs inside a Unity Test Framework Editor test assembly if you want to run them.
using NUnit.Framework;
using UnityEngine;
using UIImageStudio;

public sealed class UIStudioMathTests
{
    [Test]
    public void Snap_RoundsToNearestStep()
    {
        Assert.AreEqual(32f, UIStudioMath.Snap(30f, 16f));
        Assert.AreEqual(16f, UIStudioMath.Snap(22f, 16f));
    }

    [Test]
    public void CanvasToScreenRect_AppliesCanvasOffsetAndZoom()
    {
        Rect result = UIStudioMath.CanvasToScreenRect(
            new Rect(100f, 50f, 960f, 540f),
            new Rect(200f, 100f, 400f, 200f),
            0.5f);

        Assert.AreEqual(new Rect(200f, 100f, 200f, 100f), result);
    }

    [Test]
    public void ClampToCanvas_EnforcesMinimumSize()
    {
        Rect result = UIStudioMath.ClampToCanvas(new Rect(10f, 10f, 1f, 2f), new Vector2(1920f, 1080f));
        Assert.GreaterOrEqual(result.width, 8f);
        Assert.GreaterOrEqual(result.height, 8f);
    }
}
